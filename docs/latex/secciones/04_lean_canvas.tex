\section{Parte 3 y Trabajo Extra: Lean Canvas del SIGC-CUSCO}

\noindent\small El \textit{Lean Canvas} condensa el modelo de producto y negocio en una sola vista de nueve bloques bajo el enfoque de Ash Maurya, articulando las necesidades de las municipalidades del Cusco y del sector turismo.

\vspace{0.2cm}
\begin{center}
\setlength{\tabcolsep}{2.5pt}
\renewcommand{\arraystretch}{1}
\begin{tabularx}{\linewidth}{|Y|Y|Y|Y|Y|}
\hline
\cellcolor{CanvasLight}\CanvasCell{1. Problema}{3.3cm}{
\textbullet{} Datos dispersos en Excel, formularios, mensajer\'ia y listas f\'isicas.\par
\textbullet{} Registro de asistencia lento, vulnerable a suplantaci\'on o p\'erdida.\par
\textbullet{} Emisi\'on lenta de diplomas (semanas) y alto riesgo de falsificaci\'on.\par
\textbullet{} Falta de trazabilidad y estad\'isticas del impacto formativo.
} &
\cellcolor{CanvasLight}\CanvasCell{2. Soluci\'on}{3.3cm}{
\textbullet{} Portal web de inscripci\'on en l\'inea con validaci\'on de DNI.\par
\textbullet{} Control de asistencia digital en tiempo real v\'ia c\'odigo QR / PIN.\par
\textbullet{} Registro de evaluaciones y cierre autom\'atico de actas.\par
\textbullet{} Emisi\'on masiva de certificados PDF con firma y QR p\'ublico.
} &
\cellcolor{CanvasAccent}\CanvasCell{3. Propuesta de Valor}{3.3cm}{
\textbf{Gesti\'on de capacitaciones sin fricci\'on: de la inscripci\'on al certificado digital con QR en un solo clic.}\par
\vspace{0.3em}
Un proceso unificado, auditable y seguro para municipalidades y empresas tur\'isticas.
} &
\cellcolor{CanvasGreen}\CanvasCell{4. Ventaja Injusta}{3.3cm}{
\textbullet{} Adaptaci\'on nativa a flujos p\'ublicos peruanos (OSCE, SIGA, control municipal).\par
\textbullet{} Plantillas para sector turismo cusque\~no (sellos de calidad).\par
\textbullet{} Respaldo criptogr\'afico SHA-256 sin intermediarios costosos.
} &
\cellcolor{CanvasLight}\CanvasCell{5. Segmento Clientes}{3.3cm}{
\textbullet{} Municipalidades del Cusco (Administraci\'on, Turismo, Log\'istica).\par
\textbullet{} Empresas tur\'isticas (hoteles, agencias, restaurantes).\par
\vspace{0.2em}
\textit{Early Adopters:} Municipalidades con cursos SIGA/OSCE y hoteles en recertificaci\'on de calidad.
} \\
\hline
\cellcolor{CanvasLight}\CanvasCell{Alternativas Actuales}{2.1cm}{
Planillas Excel sueltas, Google Forms sin control de cupos, diplomas manuales en Canva o Word y listas en papel.
} &
\cellcolor{CanvasLight}\CanvasCell{6. M\'etricas Clave}{2.1cm}{
\% de asistencia efectiva; tasa de aprobaci\'on; tiempo de emisi\'on ($< 10$ s); consultas de validaci\'on QR mensuales.
} &
\cellcolor{CanvasAccent}\CanvasCell{Concepto de Alto Nivel}{2.1cm}{
\textit{Inscripci\'on \'agil estilo Eventbrite con certificaci\'on auditable de un LMS moderno.}
} &
\cellcolor{CanvasGreen}\CanvasCell{Canales}{2.1cm}{
Portales municipales, redes sociales, gremios tur\'isticos (CARTUC, DIRCETUR) y correos de confirmaci\'on.
} &
\cellcolor{CanvasLight}\CanvasCell{Adoptantes Tempranos}{2.1cm}{
Municipalidad Distrital de San Pedro (Cursos SIGA) y empresas tur\'isticas del centro hist\'orico del Cusco.
} \\
\hline
\multicolumn{2}{|p{9.4cm}|}{%
  \cellcolor{CanvasLight}\CanvasCell{7. Estructura de Costes}{2.0cm}{
  \textbullet{} Servidor cloud y base de datos relacional PostgreSQL.\par
  \textbullet{} Almacenamiento seguro de PDFs en la nube (AWS S3 / GCS).\par
  \textbullet{} Servicio de correos masivos transaccionales y soporte t\'ecnico.
  }%
} &
\multicolumn{3}{p{14.5cm}|}{%
  \cellcolor{CanvasGreen}\CanvasCell{8. Flujos de Ingresos y Retorno de Inversi\'on}{2.0cm}{
  \textbullet{} Licenciamiento institucional B2G/B2B anual seg\'un volumen de capacitaciones.\par
  \textbullet{} Paquetes por cr\'editos de certificaci\'on para eventos masivos espec\'ificos.\par
  \textbullet{} Ahorro operativo del 90\,\% en papel, tinta, carpetas y horas hombre administrativas.
  }%
} \\
\hline
\end{tabularx}
\end{center}

\vspace{0.1cm}
